// The Weeknd Programming Language

// Program start
program         = "XOTWOD" "{" { stmt } "EXODUS" "}" EOF ;

// Block of statements
block           = "{" { stmt } "}" ;

// Statements
stmt            = decl_stmt
                | assign_stmt
                | io_stmt
                | if_stmt
                | loop_stmt
                | return_stmt
                | comment ;

// Declarations (variables and functions)
decl_stmt       = "CALLOUT" type IDENT [ assign_op expr ]         // variable declaration
                | "CALLOUT" type IDENT "(" [ param_list ] ")" block ; // function declaration

param_list      = param { "," param } ;
param           = type IDENT ;

// Types
type            = "xoNum"
                | "xoDouble"
                | "xoChar"
                | "xoBool"
                | "xoString"
                | "xoArray"
                | "xoVoid" ;

// Assignment
assign_stmt     = lvalue assign_op expr ;
assign_op       = "~>" ;
lvalue          = IDENT { "[" expr "]" } ;


// Input / Output
io_stmt         = listen_stmt | echo_stmt ;
listen_stmt     = "LISTEN" "(" IDENT ")" ;
echo_stmt       = "ECHO" "(" expr ")" ;


// Return
return_stmt     = "EXODUS" [ expr ] ;


// Conditionals
if_stmt         = "tryMe" "(" cond ")" block
                  { "next" "(" cond ")" block }
                  [ "orNah" block ] ;


// Loops
loop_stmt       = "repeatAfterMe" "(" assign_stmt ";" cond ";" unary_stmt ")" block
                | "timeless" "(" cond ")" block ;

unary_stmt      = IDENT ( "rise" | "fade" ) ;

// Expressions
cond            = expr rel_op expr ;
rel_op          = "==" | "!=" | "<" | ">" | "<=" | ">=" ;

expr            = logic_expr ;
logic_expr      = [ "xoNot" ] arith_expr { ( "xoAnd" | "xoOr" ) arith_expr } ;
arith_expr      = term { ( "+" | "-" ) term } ;
term            = factor { ( "*" | "/" | "%" ) factor } ;
factor          = unary_expr | atom ;
unary_expr      = unary_op expr ;
unary_op        = "rise" | "fade" | "xoNot" ;

// Atoms
atom            = NUMBER
                | REAL
                | STRING
                | CHAR
                | "TRUE"
                | "FALSE"
                | lvalue
                | array_lit
                | "(" expr ")" ;

// Array literal
array_lit       ="[" [ array_element { "," array_element } ] "]" ;
array_element   = expr | array_lit ;


// Comments
comment         = single_comment | multi_comment ;
single_comment  = "//" { any_char - newline } newline ;
multi_comment   = "/*" { comment | any_char - "*/" } "*/" ;


EOF             = ε ;   // epsilon — oznacava "nema vise unosa" - prazana skup
