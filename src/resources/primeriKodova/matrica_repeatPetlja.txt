XOTWOD {
    CALLOUT xoArray xoArray xoNum matrix ~>
    [
        [1, 2, 3],
        [4, 5, 6],
        [7, 8, 9]
    ];

    ECHO("Elementi matrice:");

    repeatAfterMe(CALLOUT xoNum i ~> 0; i < 3; rise i) {
        repeatAfterMe(CALLOUT xoNum j ~> 0; j < 3; rise j) {
            ECHO(matrix[i][j]);
        }
    }

    EXODUS;
}
